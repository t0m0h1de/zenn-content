### ./chapters/ch02-basic-expressions/exercises/ex05-input-validation-functions/src/main/scala/Ex05.scala

```scala
def validationName(raw: String): Either[String, String] =
  val name = raw.trim
  if name.isEmpty then Left("name is empty")
  else Right(name)

def parseAge(raw: String): Either[String, Int] =
  val text = raw.trim
  if text.forall(_.isDigit) then Right(text.toInt)
  else Left("age must be integer")
  
def formatProfile(name: String, age: Int): String =
  s"name=$name, age=$age"

def buildProfile(nameRaw: String, ageRaw: String): Either[String, String] =
  for 
    name <- validationName(nameRaw)
    age <- parseAge(ageRaw)
  yield formatProfile(name, age)

@main def Ex05(): Unit =
  println(buildProfile(" Tomo ", " 30 ")) // Right(name=Tomo, age=30)
  println(buildProfile(" ", " 30 ")) // Left(name is empty)
  println(buildProfile(" Tomo ", " thirty ")) // Left(age must be integer)
```

```scala
enum OrderStatus:
  case Created, Paid, Shipped, Canneled

sealed trait PaymentResult
object PaymentResult:
  case object Success extends PaymentResult
  final case class Failed(reason: String) extends PaymentResult

case class Order(status: OrderStatus):
  def pay(): (PaymentResult, Order) =
    if status == OrderStatus.Created then
      (PaymentResult.Success, this.copy(OrderStatus.Paid))
    else 
      (PaymentResult.Failed("Order is not in a state that can be paid"), this)

def printOrderStatus(order: Order): Unit =
  println(s"Order status: ${order.status}")

@main def Ex02(): Unit =
  val order = Order(OrderStatus.Created)
  printOrderStatus(order)
  order.pay() match
    case (PaymentResult.Success, updatedOrder) =>
      println("Payment successful!")
      printOrderStatus(updatedOrder)
    case (PaymentResult.Failed(reason), _) =>
      println(s"Payment failed: $reason")

  val invalidOrder = Order(OrderStatus.Shipped)
  invalidOrder.pay() match
    case (PaymentResult.Success, updatedOrder) =>
      println("Payment successful!")
      printOrderStatus(updatedOrder)
    case (PaymentResult.Failed(reason), _) =>
      println(s"Payment failed: $reason")
```

### ./chapters/ch03-adt-case-class/exercises/ex05-order-status-adt/src/main/scala/Ex05.scala

```scala
enum OrderStatus:
  case Created, Paid, Shipped, Cancelled


def transit(current: OrderStatus, action: String): Either[String, OrderStatus] = 
  (current, action) match
    case (OrderStatus.Created, "pay") => Right(OrderStatus.Paid)
    case (OrderStatus.Paid, "ship") => Right(OrderStatus.Shipped)
    case (OrderStatus.Created, "cancel") => Right(OrderStatus.Cancelled)
    case (OrderStatus.Paid, "cancel") => Right(OrderStatus.Cancelled)
    case _ => Left("invalid transition")

@main def Ex05: Unit =
  println(transit(OrderStatus.Created, "pay")) // Right(Paid)
  println(transit(OrderStatus.Paid, "ship")) // Right(Shipped)
  println(transit(OrderStatus.Created, "cancel")) // Right(Cancelled)
  println(transit(OrderStatus.Paid, "cancel")) // Right(Cancelled)
  println(transit(OrderStatus.Shipped, "cancel")) // Left(invalid transition)
  println(transit(OrderStatus.Cancelled, "pay")) // Left(invalid transition)
```

trait UserRepo:
  def findPasswordHash(userId: String): Option[String]

  def matches(raw: String, hashed: String): Boolean

trait LoginService:
  self: UserRepo & PasswordHasher =>

  def login(userId: String, rawPassword: String): Boolean =
    findPasswordHash(userId)
      .exists { hashed =>
        matches(rawPassword, hashed)
      }

trait InMemoryUserRepo extends UserRepo:
  def findPasswordHash(userId: String): Option[String] =
    if userId == "alice" then Some("hashed:pw") else None

trait PlainHasher extends PasswordHasher:
  def matches(raw: String, hashed: String): Boolean = 
    s"hashed:$raw" == hashed

object AuthApp extends LoginService, InMemoryUserRepo, PlainHasher

trait UserGateway:
  def findHashByEmail(email: String): Option[String]

trait PasswordVerifier:
  def verify(raw: String, hashed: String): Boolean

trait TokenIssuer:
  def issue(email: String): String

trait AuthUseCase:
  self: UserGateway & PasswordVerifier & TokenIssuer =>

  def login(email: String, password: String): Either[String, String] =
    findHashByEmail(email) match
      case None => Left("user not found")
      case Some(hash) if !verify(password, hash) => Left("invalid credentials")
      case Some(_) => Right(issue(email))

trait MyUserGatewway extends UserGateway:

  def findHashByEmail(email: String): Option[String] = 
    if email == "my-user@example.com" then Some("hashed:password") else None

trait MyPasswordVerifier extends PasswordVerifier:

  def verify(raw: String, hashed: String): Boolean =
    raw == "password" && hashed == "hashed:password"

trait MyTokenIssuer extends TokenIssuer:

  def issue(email: String): String =
    s"jwt-token-for-$email"

class MyAuthUseCase extends AuthUseCase, MyUserGatewway, MyPasswordVerifier, MyTokenIssuer

@main def ex05(): Unit =
  val authUseCase = new MyAuthUseCase()

  authUseCase.login("nyanyanya", "password") match
    case Left(error) => println(s"Login failed: $error")
    case Right(token) => println(s"Login successful! Token: $token")

  authUseCase.login("my-user@example.com", "password") match
    case Left(error) => println(s"Login failed: $error")
    case Right(token) => println(s"Login successful! Token: $token")

### ./examples/patterns/src/Patterns.scala

```scala
sealed trait Expr

case class Num(n: Int) extends Expr

case class Add(left: Expr, right: Expr) extends Expr

case class Mul(left: Expr, right: Expr) extends Expr


object Pattern {
  def eval(e: Expr): Int = e match {
    case Num(n) => n
    case Add(l, r) => eval(l) + eval(r) 
    case Mul(l, r) => eval(l) * eval(r)
    }

  def describe(e: Expr): String = e match {
    case Num(n) => s"num($n)"
    case Add(l, r) => s"add({${describe(l)},{${describe(r)})"
    case Mul(l, r) => s"mul({${describe(l)},{${describe(r)})"
    }

  def main(args: Array[String]): Unit = {
    val expr = Add(Mul(Num(2), Num(3)), Num(4))
    println(describe(expr))
    val list: List[Expr] = List(Num(1), Add(Num(2), Num(3)), Num(-5))
    val nums = list.collect { case Num(n) => n }
    println(nums)

    val n = Num(5)
    println(n.copy(n = 6))
    }
  }
```
